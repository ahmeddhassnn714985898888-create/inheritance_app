class DatabaseHelper {
  static final DatabaseHelper instance = DatabaseHelper._init();
  static Database? _database;

  DatabaseHelper._init();

  Future<Database> get database async {
    if (_database != null) return _database!;
    _database = await _initDB('inheritance_app.db');
    return _database!;
  }

  Future<Database> _initDB(String filePath) async {
    final dbPath = await getDatabasesPath();
    final path = p.join(dbPath, filePath);

    return await openDatabase(
      path,
      version: 1,
      onCreate: _createDB,
    );
  }

  Future _createDB(Database db, int version) async {
    await db.execute('''
      CREATE TABLE heirs (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        relation TEXT NOT NULL,
        custom_deduction REAL DEFAULT 0.0
      )
    ''');

    await db.execute('''
      CREATE TABLE properties (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT NOT NULL,
        type TEXT NOT NULL,
        tenant_name TEXT,
        monthly_rent REAL DEFAULT 0.0
      )
    ''');

    await db.execute('''
      CREATE TABLE transactions (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        heir_id INTEGER NOT NULL,
        amount REAL NOT NULL,
        type TEXT NOT NULL,
        notes TEXT,
        date TEXT NOT NULL
      )
    ''');
  }

  Future<String> exportBackup() async {
    final db = await database;
    final heirs = await db.query('heirs');
    final properties = await db.query('properties');
    final transactions = await db.query('transactions');

    final backupData = {
      'heirs': heirs,
      'properties': properties,
      'transactions': transactions,
      'date': DateTime.now().toIso8601String(),
    };

    final directory = await getApplicationDocumentsDirectory();
    final filePath = p.join(directory.path, 'inheritance_backup.json');
    final file = File(filePath);
    await file.writeAsString(jsonEncode(backupData));
    return filePath;
  }
}

class HomeScreen extends StatefulWidget {
